cmake_minimum_required(VERSION 3.16)
project(damvi_vesc_sim C)
add_library(damvi_vesc SHARED native/vesc_speed.c native/drivetrain.c)
set_target_properties(damvi_vesc PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
target_compile_options(damvi_vesc PRIVATE -Wall -Wextra -Werror)
target_link_libraries(damvi_vesc PRIVATE m)

set_target_properties(damvi_steering PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
target_compile_options(damvi_steering PRIVATE -Wall -Wextra -Werror)
target_link_libraries(damvi_steering PRIVATE m)
